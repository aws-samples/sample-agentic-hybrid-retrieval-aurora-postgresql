EXPLAIN THE RANKING · AURORA POSTGRESQL
WITH lexical AS (
  SELECT chunk_id, ROW_NUMBER() OVER (ORDER BY ts_rank_cd(tsv, q) DESC) AS r
  FROM object_chunks, websearch_to_tsquery('orion delay decision commitments') q
  WHERE tsv @@ q AND project = 'orion' AND created_at > now() - INTERVAL '90 days'
  LIMIT 60
), semantic AS (
  SELECT chunk_id, ROW_NUMBER() OVER (ORDER BY embedding <=> :query_vec) AS r
  FROM object_chunks WHERE project = 'orion'          -- HNSW index scan
  ORDER BY embedding <=> :query_vec LIMIT 60
), fuzzy AS (
  SELECT chunk_id, ROW_NUMBER() OVER (ORDER BY similarity(title, 'ORION-1473') DESC) AS r
  FROM source_objects WHERE title % 'ORION-1473'       -- pg_trgm entity match
  LIMIT 40
)
SELECT chunk_id, SUM(w / (60 + r)) AS rrf_score        -- RRF · k = 60 · w = 1/1/0.5
FROM (
  SELECT chunk_id, r, 1.0 AS w FROM lexical
  UNION ALL SELECT chunk_id, r, 1.0 FROM semantic
  UNION ALL SELECT chunk_id, r, 0.5 FROM fuzzy
) fused
GROUP BY chunk_id ORDER BY rrf_score DESC LIMIT 24;   -- → cross-encoder rerank
